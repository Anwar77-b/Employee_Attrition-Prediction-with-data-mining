#!/usr/bin/env bash
# Creates the Data Mining mini-project structure (Employee Attrition).
# Usage: bash setup_project.sh [project_name]

set -e

ROOT="${1:-employee-attrition-mining}"

# --- Folders ---------------------------------------------------------------
mkdir -p "$ROOT"/{data/raw,data/processed,notebooks,src,models,results,scripts,tests,docs/figures}

# --- Notebooks -------------------------------------------------------------
touch "$ROOT"/notebooks/01_eda_and_cleaning.ipynb \
      "$ROOT"/notebooks/02_statistical_analysis.ipynb \
      "$ROOT"/notebooks/03_preprocessing.ipynb \
      "$ROOT"/notebooks/04_model_training_evaluation.ipynb

# --- Source code -----------------------------------------------------------
touch "$ROOT"/src/__init__.py \
      "$ROOT"/src/app.py \
      "$ROOT"/src/preprocessing.py \
      "$ROOT"/src/models.py \
      "$ROOT"/src/evaluation.py \
      "$ROOT"/src/statistics.py \
      "$ROOT"/src/utils.py

# --- Scripts & tests -------------------------------------------------------
touch "$ROOT"/scripts/download_data.py \
      "$ROOT"/scripts/train_pipeline.py \
      "$ROOT"/tests/test_data.py \
      "$ROOT"/tests/test_preprocessing.py \
      "$ROOT"/tests/test_models.py

# --- Keep empty folders in Git ---------------------------------------------
touch "$ROOT"/data/raw/.gitkeep \
      "$ROOT"/data/processed/.gitkeep \
      "$ROOT"/models/.gitkeep \
      "$ROOT"/results/.gitkeep \
      "$ROOT"/docs/figures/.gitkeep

# --- Top-level files -------------------------------------------------------
touch "$ROOT"/README.md "$ROOT"/LICENSE

cat > "$ROOT"/requirements.txt <<'EOF'
pandas
numpy
scipy
scikit-learn
statsmodels
matplotlib
seaborn
plotly
streamlit
joblib
kagglehub
pytest
jupyter
